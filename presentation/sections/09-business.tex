\begin{frame}{Business model: subscription}
  \adminikicker{recurring revenue}
  \vspace{1.5mm}

  \begin{claimcard}{supported}{personal}
    Monthly or annual. Unlimited checking, history and local mode.
  \end{claimcard}
  \begin{claimcard}{supported}{teams}
    Per-seat subscription. Shared, traceable, exportable checks.
  \end{claimcard}
  \begin{claimcard}{doubtful}{institutions}
    Annual enterprise subscription. Audit logs and reporting API.
  \end{claimcard}

  \adminisource{Recurring revenue; a few cents a check, removed by a local model.}
  \note{%
    One model, three tiers. Personal is a monthly or annual subscription for
    unlimited checking, saved history and a private local mode. Teams---newsrooms
    and schools---pay per seat, because a check there is part of a workflow and
    has to be traceable and exportable. Institutions pay an annual enterprise
    subscription for audit-grade check logs and an API that maps to DSA and AI
    Act reporting duties. Subscription smooths the variable cost; a local model
    raises the margin. Price is anchored to the cost of a human verification
    hour, not to tokens.%
  }
\end{frame}
