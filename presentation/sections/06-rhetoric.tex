\begin{frame}{How it is argued, not just whether it is true}
  \adminikicker{innovation \textperiodcentered\ rhetorical pass}
  \vspace{2mm}

  \adminilead{A claim can be accurate and still be deployed to mislead---a true
    figure with its base rate removed. That reading runs beside fact-checking,
    never inside it.}

  \vspace{3mm}
  \administat{8}{techniques}
  \administat{0/0}{neutral opinion}
  \administat{2/2}{same manipulation}

  \vspace{3mm}
  \begin{itemize}
    \item Each signal must quote a verbatim transcript line or it is dropped.
    \item The prompt names the technique, never the position.
  \end{itemize}
  \note{%
    The mirror test: 0/0 on neutral political opinion from either side, 2/2 on
    the same manipulation from either side. Offer the caveat yourself:
    constructed examples, real speech is subtler, we would not overclaim. An
    empty list is the normal, correct answer for an ordinary video. This is the
    other half of the 30\% for innovation.%
  }
\end{frame}
