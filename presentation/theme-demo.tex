\begin{frame}{Verdict rules}

  \end{claimcard}
  \begin{claimcard}{doubtful}{07:41}
    A claim that survives one source but not a second.
  \end{claimcard}
  \begin{claimcard}{misleading}{11:05}

  \end{claimcard}
\end{frame}
